MIDDLEWARE EnsureUserHasDepartamento:
   public function handle(Request $req, Closure $next)
   {
       $user = $req->user();
       if (! $user) return redirect()->route('login');

       $rutasPermitidas = ['inicio','perfil.show','perfil.update','logout'];
       if (in_array($req->route()?->getName(), $rutasPermitidas)) {
           return $next($req);
       }

       if ($user->hasRole('superadmin')) {
           return $next($req); // bypass
       }

       if (! $user->departamento_id || ! $user->nivel_acceso_id) {
           if ($req->expectsJson()) {
               return response()->json([
                   'message' => 'Sin departamento asignado',
                   'code'    => 'NO_DEPARTAMENTO',
               ], 403);
           }
           return redirect()->route('inicio')->with('warning',
               'Tu cuenta no tiene departamento asignado. '.
               'Contacta al administrador.');
       }

       return $next($req);
   }
